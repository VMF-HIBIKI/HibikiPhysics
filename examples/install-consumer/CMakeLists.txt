cmake_minimum_required(VERSION 4.4)
project(HibikiPhysicsConsumer LANGUAGES CXX)
find_package(HibikiPhysics CONFIG REQUIRED)
add_executable(hibiki_consumer main.cpp)
target_link_libraries(hibiki_consumer PRIVATE hibiki::physics)
enable_testing()
add_test(NAME consumer.version COMMAND hibiki_consumer)

add_executable(hibiki_plugin_consumer PluginConsumer.cpp)
target_link_libraries(hibiki_plugin_consumer PRIVATE hibiki::physics)
add_test(NAME consumer.plugin COMMAND hibiki_plugin_consumer)

add_executable(hibiki_scene_consumer SceneConsumer.cpp)
target_link_libraries(hibiki_scene_consumer PRIVATE hibiki::physics)
add_test(NAME consumer.scene COMMAND hibiki_scene_consumer "${CMAKE_CURRENT_SOURCE_DIR}/sphere-plane.hscene")
